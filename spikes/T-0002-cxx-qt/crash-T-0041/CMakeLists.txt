# T-0041 reproducer: C++-only harness around the T-0002 async image provider (several provider variants).
# Debug build uses Qt's debug DLLs (Qt6Cored etc.); -DTW_ASAN=ON adds MSVC AddressSanitizer.
cmake_minimum_required(VERSION 3.24)
project(t0041_repro LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(repro main.cpp providers.cpp)
target_link_libraries(repro PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick dbghelp)
if(MSVC)
  target_compile_options(repro PRIVATE /Zi /utf-8)
  target_link_options(repro PRIVATE /DEBUG /INCREMENTAL:NO)
  if(TW_ASAN)
    target_compile_options(repro PRIVATE /fsanitize=address)
  endif()
endif()

# Minimal QtCore-only model of the first shim's cross-thread emit (see race.cpp)
add_executable(race race.cpp)
target_link_libraries(race PRIVATE Qt6::Core dbghelp)
if(MSVC)
  target_compile_options(race PRIVATE /Zi /utf-8)
  target_link_options(race PRIVATE /DEBUG /INCREMENTAL:NO)
endif()
